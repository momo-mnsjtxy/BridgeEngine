cmake_minimum_required(VERSION 3.20)
project(BridgeEnginePackageConsumer LANGUAGES C CXX)

enable_testing()

if(NOT DEFINED EXPECT_SHARED OR NOT DEFINED EXPECT_STATIC)
	message(FATAL_ERROR "EXPECT_SHARED and EXPECT_STATIC are required")
endif()

find_package(BridgeEngine CONFIG REQUIRED)

function(bridgeengine_assert_target target_name expected type)
	if(expected)
		if(NOT TARGET ${target_name})
			message(FATAL_ERROR "Expected exported target ${target_name} is missing")
		endif()
		get_target_property(actual_type ${target_name} TYPE)
		if(NOT actual_type STREQUAL type)
			message(FATAL_ERROR "${target_name} has type ${actual_type}, expected ${type}")
		endif()
	elseif(TARGET ${target_name})
		message(FATAL_ERROR "Unexpected exported target ${target_name}")
	endif()
endfunction()

bridgeengine_assert_target(BridgeEngine::bridgeengine_shared ${EXPECT_SHARED} SHARED_LIBRARY)
bridgeengine_assert_target(BridgeEngine::bridgeengine_static ${EXPECT_STATIC} STATIC_LIBRARY)

get_property(imported_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
set(actual_product_targets)
foreach(target_name IN LISTS imported_targets)
	if(target_name MATCHES "^BridgeEngine::bridgeengine_")
		list(APPEND actual_product_targets ${target_name})
	endif()
endforeach()
set(expected_product_targets)
if(EXPECT_SHARED)
	list(APPEND expected_product_targets BridgeEngine::bridgeengine_shared)
endif()
if(EXPECT_STATIC)
	list(APPEND expected_product_targets BridgeEngine::bridgeengine_static)
endif()
list(SORT actual_product_targets)
list(SORT expected_product_targets)
if(NOT actual_product_targets STREQUAL expected_product_targets)
	message(FATAL_ERROR "Exported product targets ${actual_product_targets} do not match ${expected_product_targets}")
endif()

function(bridgeengine_add_compatibility_consumers variant target_name)
	foreach(language IN ITEMS c cpp)
		add_executable(consumer_${variant}_${language}
			"${CMAKE_CURRENT_LIST_DIR}/../master_aggregate_bapi_compat_test.${language}")
		target_link_libraries(consumer_${variant}_${language} PRIVATE ${target_name})
		add_test(NAME consumer.${variant}.${language} COMMAND consumer_${variant}_${language})
	endforeach()
endfunction()

if(EXPECT_SHARED)
	bridgeengine_add_compatibility_consumers(shared BridgeEngine::bridgeengine_shared)
endif()
if(EXPECT_STATIC)
	bridgeengine_add_compatibility_consumers(static BridgeEngine::bridgeengine_static)
endif()
