cmake_minimum_required(VERSION 3.20)
project(BridgeEnginePkgConfigStaticFixture LANGUAGES C)

if(USE_EXISTING_SDL3)
	add_library(PkgConfig::SDL3 INTERFACE IMPORTED)
	set_target_properties(PkgConfig::SDL3 PROPERTIES
		INTERFACE_COMPILE_DEFINITIONS BRIDGEENGINE_EXISTING_SDL3
	)
endif()

find_package(BridgeEngine CONFIG REQUIRED)

get_target_property(sdl3_libraries PkgConfig::SDL3 INTERFACE_LINK_LIBRARIES)
get_target_property(sdl3_directories PkgConfig::SDL3 INTERFACE_LINK_DIRECTORIES)
get_target_property(sdl3_link_options PkgConfig::SDL3 INTERFACE_LINK_OPTIONS)
get_target_property(sdl3_compile_options PkgConfig::SDL3 INTERFACE_COMPILE_OPTIONS)
get_target_property(ffmpeg_libraries PkgConfig::FFMPEG INTERFACE_LINK_LIBRARIES)
if(USE_EXISTING_SDL3)
	get_target_property(sdl3_definitions PkgConfig::SDL3 INTERFACE_COMPILE_DEFINITIONS)
	if(NOT sdl3_definitions STREQUAL "BRIDGEENGINE_EXISTING_SDL3")
		message(FATAL_ERROR "existing PkgConfig::SDL3 target was modified")
	endif()
else()
	foreach(expected fixture_sdl3 fixture_sdl3_private)
		if(NOT expected IN_LIST sdl3_libraries)
			message(FATAL_ERROR "PkgConfig::SDL3 is missing ${expected}")
		endif()
	endforeach()
	if(NOT "/tmp" IN_LIST sdl3_directories)
		message(FATAL_ERROR "PkgConfig::SDL3 is missing its fixture link directory")
	endif()
	if(NOT "-pthread" IN_LIST sdl3_link_options OR NOT "-Wl,--as-needed" IN_LIST sdl3_link_options)
		message(FATAL_ERROR "PkgConfig::SDL3 is missing fixture link options")
	endif()
	if(NOT "-DFIXTURE_SDL3" IN_LIST sdl3_compile_options)
		message(FATAL_ERROR "PkgConfig::SDL3 is missing fixture compile options")
	endif()
	if(APPLE)
		if(NOT "SHELL:-framework FixtureFramework" IN_LIST sdl3_link_options)
			message(FATAL_ERROR "PkgConfig::SDL3 did not preserve the fixture framework option")
		endif()
	else()
		if(NOT "-framework" IN_LIST sdl3_link_options OR NOT "FixtureFramework" IN_LIST sdl3_link_options)
			message(FATAL_ERROR "PkgConfig::SDL3 is missing the fixture framework option")
		endif()
	endif()
	if(NOT "fixture_avcodec" IN_LIST ffmpeg_libraries)
		message(FATAL_ERROR "PkgConfig::FFMPEG is missing fixture libraries")
	endif()
	if("${sdl3_libraries};${ffmpeg_libraries}" MATCHES "/usr")
		message(FATAL_ERROR "PkgConfig static targets leaked host paths")
	endif()
	add_library(pkgconfig_compile_probe OBJECT probe.c)
	target_link_libraries(pkgconfig_compile_probe PRIVATE BridgeEngine::SDL3)
endif()
